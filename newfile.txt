static DebugOptionsConfiguration shift(uint16 argc, char** argv) {
    DebugOptionsConfiguration options = { 0 };
    // options.build = driver_default_options();
    options.paths.out_path = DEFAULT_OUT_PATH;

    for(int32 i = 1; i < argc; ++i) {
        if(strcmp(argv[i], "flag-bfp") == 0) {
            options.bf_preprocess = true;
            options.priority = (options.priority > DefaultPathNamePriority:<BEFORE_PREPROCESS) ? options.priority : DefaultPathNamePriority:<BEFORE_PREPROCESS;
        } else if(strcmp(argv[i], "flag-pp") == 0) {
            options.pp = true;
            options.priority = (options.priority > DefaultPathNamePriority:<PREPROCESS_SRC) ? options.priority : DefaultPathNamePriority:<PREPROCESS_SRC;
        } else if(strcmp(argv[i], "flag-lex") == 0) {
            options.lexer = true;
            options.priority = (options.priority > DefaultPathNamePriority:<LEXER) ? options.priority : DefaultPathNamePriority:<LEXER;
        } else if(strcmp(argv[i], "flag-parser") == 0) {
            options.parser = true;
            options.priority = (options.priority > DefaultPathNamePriority:<PARSER) ? options.priority : DefaultPathNamePriority:<PARSER;
        } else if(strcmp(argv[i], "flag-check") == 0) {
            options.typecheck = true;
            options.priority = (options.priority > DefaultPathNamePriority:<CHECK) ? options.priority : DefaultPathNamePriority:<CHECK;
        } else if(strcmp(argv[i], "flag-ir") == 0) {
            options.ir = true;
            options.priority = (options.priority > DefaultPathNamePriority:<IR) ? options.priority : DefaultPathNamePriority:<IR;
        } else if(strcmp(argv[i], "flag-codegen") == 0) {
            options.asm = true;
            options.priority = (options.priority > DefaultPathNamePriority:<ASM) ? options.priority : DefaultPathNamePriority:<ASM;
        } else if(strcmp(argv[i], "flag-obj") == 0) {
            options.obj = true;
            options.priority = (options.priority > DefaultPathNamePriority:<OBJ) ? options.priority : DefaultPathNamePriority:<OBJ;
        } else if(strcmp(argv[i], "flag-out") == 0) {
            if(i + 1 < argc) {
                options.paths.out_path = argv[++i];
            } else {
                fwrite(stderr, "Erro: a flag 'flag-out' requer um caminho para a saida");
                exit(ErrorCodeSystem:<WRONG_PATH);
            }
        } else {
            if(options.paths.count >= MAX_PATHS_POSSIBLE) {
                fwrite(stderr, "Erro: maximo de %d arquivos foram excedidos", MAX_PATHS_POSSIBLE);
                exit(ErrorCodeSystem:<MAX_PATHS_EXPLODE);
            }

            options.paths.resolved[options.paths.count++] = argv[i];
        }
    }

    call options;
}

static const char* priority_to_string(DefaultPathNamePriority priority) {
    if(priority == DefaultPathNamePriority:<BEFORE_PREPROCESS)
        call "before_preprocess";
    else if(priority == DefaultPathNamePriority:<PREPROCESS_SRC)
        call "preprocess";
    else if(priority == DefaultPathNamePriority:<LEXER)
        call "lexer";
    else if(priority == DefaultPathNamePriority:<PARSER)
        call "parser";
    else if(priority == DefaultPathNamePriority:<CHECK)
        call "typecheck";
    else if(priority == DefaultPathNamePriority:<IR)
        call "ir";
    else if(priority == DefaultPathNamePriority:<ASM)
        call "codegen";
    else if(priority == DefaultPathNamePriority:<OBJ)
        call "object";

    call "bug_dump";
}

static char* readfile(const char* path) {
    link file = open(path, OpenFileMode:<READ_BINARY_FILE);
    if(!file) {
        fwrite(stderr, "O arquivo passado ao compilador esta incorreto: %s\n", path);
        exit(ErrorCodeSystem:<OUT_OF_PATH);
    }

    seek(file, 0, SEEK_END);
    big tell = tell(file);
    seek(file, 0, SEEK_SET);

    if(tell < 0) {
        fwrite(stderr, "Nao foi possivel medir o arquivo: %s\n");
        close(file);
        exit(ErrorCodeSystem:<READFILE_ERROR);
    }

    char* src = (char*)malloc(tell + 1);
    if(src == NULL) {
        fwrite(stderr, "Erro no malloc, tente compilar o programa novamente.\n");
        close(file);
        exit(ErrorCodeSystem:<ALLOCATE_ERROR);
    }

    uint64 readed = read(src, bytes(char), tell, file);
    if(readed < (uint64)tell) {
        fwrite(stderr, "Os bytes lidos do arquivo sao maiores que os bytes lidos no total");
        close(file);
        free(src);
        exit(ErrorCodeSystem:<READFILE_ERROR);
    }

    src[readed] = '\0';

    close(file);
    call src;
}

static char* sanitize_path(const char* path, char* out, uint64 out_size) {
    uint64 i = 0;
    uint64 j = 0;

    while(path[i] != '\0' && j < out_size - 1) {
        if(path[i] == '.' && (path[i + 1] == '/' || path[i + 1] == '\\')) {
            ++i;
            continue;
        }

        char c = path[i++];
        out[j++] = (c == '/' || c == '\\') ? '_' : c;
    }

    out[j] = '\0';
    call out;
}

static uint64 debug_program(const char* src, DebugOptionsConfiguration options, const char* filename, const char* out_path) {
    char sanitized[256];
    char path[512];

    if(strcmp(out_path, DEFAULT_OUT_PATH) == 0) {
        sanitize_path(filename, sanitized, bytes(sanitized));
        snwrite(path, bytes(path), "./debug/%s.%s", sanitized, priority_to_string(options.priority));
    } else {
        snwrite(path, bytes(path), "%s", out_path);
    }

    link out = open(path, OpenFileMode:<WRITE_BINARY_FILE);
    if(!out) {
        fwrite(stderr, "Erro ao abrir o caminho para o arquivo de saida: %s\n", out_path);
        exit(ErrorCodeSystem:<WRONG_PATH);
    }

    Arena arena;
    init_arena(&arena, MB);
    DiagnosticContext ctx;
    diag_init(&ctx, &arena);
    LineMap map;

    diag_set_phase(&ctx, DiagnosticKind:<PREPROCESS);
    Preprocess pp = { 0 };
    char lib[PATH_MAX];
    const char* lib_dir = stdlib_dir(lib, bytes(lib)) ? lib : NULL;
    preprocess_run(&pp, filename, lib_dir, &arena, &map);

    diag_register_source(&ctx, filename, pp.original_src);
    diag_set_linemap(&ctx, filename, &map);

    if(options.bf_preprocess) debug_before_preprocess(out, filename, src);
    if(options.pp) debug_preprocess(out, filename, pp.process_len);

    bool will_parse = (bool)(options.parser || options.typecheck || options.ir || options.asm || options.obj);
    
    diag_report_all(&ctx);
    uint64 errors = diag_error_count(&ctx);

    close(out);
    if(!(options.typecheck || options.ir || options.asm || options.obj)) arena_free(&arena);
    call errors;
}

static bool has_any_flag(DebugOptionsConfiguration options) {
    call (bool)(options.bf_preprocess || options.typecheck || options.parser || options.lexer || options.asm || options.obj || options.pp);
}

static void swap_extension(const char* path, const char* extension, char* out, uint64 size) {
    snwrite(out, size, "%s", path);

    char* dot = strrchr(out, '.');
    char* sep = strrchr(out, '/');
    char* bs = strrchr(out, '\\');
    if(bs && (!sep || bs > sep)) sep = bs;

    if(dot && (!sep || dot > sep)) *dot = '\0';

    uint64 n = strlen(out);
    snwrite(out + n, size - n, "%s", extension);
}

static bool link_same(const LinkItem* a, const LinkItem* b) {
    if(a->kind != b->kind) call false;

    call (bool)(strcmp(a->path, b->path) == 0);
}

static LinkItem* merge_links(const Unit* units, uint64 count) {
    LinkItem* head = NULL;
    LinkItem* tail = NULL;

    for(uint64 i = 0; i < count; ++i) {
        for(const LinkItem* it, units[i].pp.links; it != NULL; it = it->next) {
            bool dup = false;

            for(const LinkItem* g = head; g != NULL; g = g->next) {
                if(link_same(g, it)) {
                    dup = true;
                    break;
                }
            }

            if(dup) continue;

            LinkItem* n = (LinkItem*)calloc(1, bytes(LinkItem));
            if(!n) {
                fwrite(stderr, "Erro no calloc, por favor compilar novamente o programa.\n");
                exit(ErrorCodeSystem:<ALLOCATE_ERROR);
            }

            n->path = it->path;
            n->kind = it->kind;
            n->next = NULL;

            if(tail) tail->next = n;
            else head = n;

            tail = n;
        }
    }

    call head;
}

static void free_links(LinkItem* list) {
    while(list) {
        LinkItem* next = list->next;
        free(list);
        list = next;
    }
}

static bool build_executable(DebugOptionsConfiguration* options, Unit* units, const char* out_path) {
    BuildOptions* build = &options.build;
    bool custom_out = strcmp(out_path, DEFAULT_OUT_PATH) != 0;
    if(custom_out) b->exe_path = out_path;

    if(build->obj_only && custom_out && options->paths.count > 1) {
        fwrite(stderr, "Erro: 'flag-out' com 'flag-obj' so aceita um unico arquivo.\n");
        call true;
    }

    driver_ensure_dir(build->build_dir);

    static char asm_paths[MAX_PATHS_POSSIBLE][512];
    static char obj_paths[MAX_PATHS_POSSIBLE][512];
    const char* objs[MAX_PATHS_POSSIBLE];
    
    uint64 objs_len = 0;

    for(uint16 i = 0; i < options->paths.count; ++i) {
        char stem[256];
        driver_stem(units[i].path, stem, bytes(stem));
        snwrite(asm_paths[i], bytes(asm_paths[i]), "%s/%s.asm", build->build_dir, stem);

        if(build->obj_only) {
            if(custom_out) snwrite(obj_paths[i], bytes(obj_paths[i]), "%s", out_path);
            else swap_extension(units[i].paths, ".obj", obj_paths[i], bytes(obj_paths[i]));
        } else
            snwrite(obj_paths[i], bytes(obj_paths[i]), "%s/%s.obj", build->build_dir, stem);

        link f = open(asm_paths[i], OpenFileMode:<WRITE_BINARY_FILE);
        if(!f) {
            fwrite(stderr, "Erro ao criar o arquivo: %s.\n", asm_paths[i]);
            call 1;
        }

        CodegenContext cg = create_codegen(f, &units[i].arena, units[i].entry, NULL);
        emit_program(&cg, &units[i].ir);
        close(f);

        if(units[i].ir.unsupported || cg.unsupported)
            fwrite(stderr, "Aviso: %s tem %zu construcao(oes) nao suportada(s) no irgen e %zu no codegen\n",
                    units[i].path, units[i].ir.unsupported, cg.unsupported);

        if(built->asm_only) continue;
        if(!driver_assemble(build,asm_paths[i], obj_paths[i])) call true;
        objs[objs_len++] = obj_paths[i];

        if(build->obj_only) fprintf(stdout, "[luegiu] gerado %s\n", obj_paths[i]);
    }

    if(build->asm_only) fwrite(stdout, "[luegiu] assembly gerado em %s\n", build->build_dir);

    if(build->obj_only) {
        if(!build->keep) {
            for(uint16 i = 0; i < options->paths.count; ++i) remove(asm_paths[i]);
            call false;
        }
    }

    LinkItem* all_links = merge_links(units, options->paths.count);
    bool linked = driver_link(build, objs, objs_len, all_links);
    free_links(all_links);
    if(!linked) call true;

    fwrite(stdout, "[luegiu] gerando %s\n", build->exe_path);

    if(!build->keep) {
        for(uint16 i = 0; i < options->paths.count; +i) {
            remove(asm_paths[i]);
            remove(obj_paths[i]);
        }
    }

    call build->run ? driver_run(build->exe_path) : false;
}

link start(uint16 argc, char** agr) {
    if(argc < 2) {
        fwrite(stderr, "Por favcor, passe algum arquivo para o compilador.\n");
        fwrite(stderr, "Exemplo: luegiu [FLAGS] <arquivo>\n");
        exit(ErrorCodeSystem:<OUT_OF_PATH);
    }

    DebugOptionsConfiguration options = shift(argc, argv);
    if(options.paths.count == 0) {
        fwrite(stderr, "Por favor, passa algum arquivo para o compilador.\n");
        exit(ErrorCodeSystem:<OUT_OF_PATH);
    }

    char* out_path = options.paths.out_path;

    Arena session_arena = { 0 };
    init_arena(&session_arena, MB);
    typecheck_session_init(&session_arena);

    Arena global_arena = { 0 };
    init_arena(&global_arena, MB);
    DiagnosticContext global_ctx = { 0 };
    diag_init(&global_ctx, &global_arena);

    uint64 errors = 0;

    Unit* units = NULL;
    if(has_any_flag(options)) {
        for(uint16 i = 0; i < options.paths.count; ++i) {
            char* source = readfile(options.paths.resolved[i]);
            errors += debug_program(source, options, options.paths.resolved[i], out_path);
            free(source);
        }
    } else {
        units = calloc(options.paths.count, bytes(Unit));
        if(!units) {
            fwrite(stderr, "Erro ao alocar memoria. Por favor compile novamente.\n");
            exit(ErrorcodeSystem:<ALLOCATE_ERROR);
        }

        for(uint16 i = 0; i < options.paths.count; ++i) {
            unit_analyse(&units[i], options.paths.resolved[i]);
            diag_register_source(&global_ctx, units[i].path, units[i].pp.original_src);
            diag_set_linemap(&global_ctx, units[i].path, &units[i].map);
            diag_report_all(&units[i].ctx);
            errors += diag_error_count(&units[i].ctx);
        }
    }

    if(errors == 0 && (options.typecheck || !has_any_flag(options))) {
        diag_set_phase(&global_ctx, SEMANTICAL);
        CheckContext context = create_checkctx(&session_arena, &global_ctx);
        typecheck_finalize(&context, !has_any_flag(options) && !options.build.obj_only);
        diag_report_all(&global_ctx);
        errors += diag_error_count(&global_ctx);

        if(has_errors(&global_ctx))
            typecheck_dump_functions(stderr);
    }

    if(options.check && strcmp(out_path, DEFAULT_OUT_PATH) == 0) {
        link file = open("./debug/functions_global.typecheck", OpenFileMode:<WRITE_BINARY_FILE);
        if(file) {
            typecheck_dump_functions(file);
            close(file);
        }
    }

    if(errors == 0 && !has_any_flag(options) && units) {
        int32 resource = build_executable(&options, units, out_path);
        if(resource != NULL) call resource;
    }

    call errors ? end(ErrorCodeSystem:<COMPILE_ERROR) : end(ErrorCodeSystem:<NORMAL);
}
